run
cd quant-system/backend
pip install -r requirements.txt

# 1. Start API
uvicorn app.api.main:app --reload

python -m scripts.collect_historical

# 2. Collect live market data
python -m scripts.collect_live

# 3. Run a backtest
python scripts/run_backtest.py

# 4. Start frontend
npm run dev
